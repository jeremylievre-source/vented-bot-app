☁️ Architecture 100% Cloud
Ton bot tourne 24/7 dans le cloud, même téléphone éteint.

Stack gratuite :
Cloudflare Workers → scraping Vinted (0$/mois)
Telegram Bot API → envoi des deals automatique
Cron Triggers → scan à 10h, 15h, 20h
🚀 Déploiement en 5 min
1
Créer un compte Cloudflare (gratuit) sur cloudflare.com
2
Installer Wrangler CLI sur ton Mac une seule fois :
npm install -g wrangler
3
Login Cloudflare depuis le terminal :
wrangler login
4
Copier le Worker ci-dessous dans un fichier worker.js et déployer :
wrangler deploy worker.js --name drip-deals
5
Ajouter ton TELEGRAM_TOKEN et CHAT_ID en secrets Cloudflare. C'est tout !
wrangler secret put TELEGRAM_TOKEN wrangler secret put CHAT_ID
📄 Code Worker Cloudflare
Colle ce code dans worker.js :
// DRIP.DEALS - Cloudflare Worker // Scan Vinted 20 pays toutes les 3h const VINTED_COUNTRIES = [ { code: 'fr', domain: 'www.vinted.fr' }, { code: 'de', domain: 'www.vinted.de' }, { code: 'be', domain: 'www.vinted.be' }, { code: 'nl', domain: 'www.vinted.nl' }, { code: 'es', domain: 'www.vinted.es' }, { code: 'it', domain: 'www.vinted.it' }, { code: 'pl', domain: 'www.vinted.pl' }, { code: 'cz', domain: 'www.vinted.cz' }, { code: 'sk', domain: 'www.vinted.sk' }, { code: 'hu', domain: 'www.vinted.hu' }, { code: 'ro', domain: 'www.vinted.ro' }, { code: 'lt', domain: 'www.vinted.lt' }, { code: 'lv', domain: 'www.vinted.lv' }, { code: 'fi', domain: 'www.vinted.fi' }, { code: 'se', domain: 'www.vinted.se' }, { code: 'at', domain: 'www.vinted.at' }, { code: 'pt', domain: 'www.vinted.pt' }, { code: 'lu', domain: 'www.vinted.lu' }, { code: 'uk', domain: 'www.vinted.co.uk' }, { code: 'us', domain: 'www.vinted.com' } ]; // Mots-clés multilingues pour "Perle Rare" const RARE_KEYWORDS = [ 'vintage','rare','limited','deadstock','90s','80s','70s', 'selten','limitiert','raras','raro','rara','rzadki','ritka', 'vintage','ancien','collector','og','archive' ]; async function searchVinted(domain, brand, priceMin, priceMax, rareMode) { const query = encodeURIComponent(brand); const url = \`https://\${domain}/api/v2/catalog/items?search_text=\${query}&price_from=\${priceMin}&price_to=\${priceMax}&per_page=20\`; try { const res = await fetch(url, { headers: { 'User-Agent': 'Mozilla/5.0 (iPhone; CPU iPhone OS 16_0 like Mac OS X)', 'Accept': 'application/json', 'Cookie': 'anon_id=vinted_worker_1' } }); if (!res.ok) return []; const data = await res.json(); let items = data.items || []; if (rareMode) { items = items.filter(item => { const t = (item.title || '').toLowerCase(); const d = (item.description || '').toLowerCase(); return RARE_KEYWORDS.some(k => t.includes(k) || d.includes(k)); }); } return items.map(item => ({ title: item.title, brand: brand, price: parseFloat(item.price?.amount || item.price || 0), image: item.photos?.[0]?.url || null, url: \`https://\${domain}/items/\${item.id}\`, country: domain, rare: rareMode && RARE_KEYWORDS.some(k => (item.title||'').toLowerCase().includes(k)) })); } catch(e) { return []; } } async function sendTelegram(token, chatId, deals) { if (!deals.length) return; const top = deals.slice(0,5); for (const deal of top) { const roi = Math.round(((deal.price * 2.8 - deal.price) / deal.price) * 100); const msg = \`\${deal.rare ? '💎 PERLE RARE' : '🔥 DEAL'}\n\n\` + \`👕 \${deal.title}\n\` + \`💰 \${deal.price}€ → ~\${Math.round(deal.price*2.8)}€\n\` + \`📈 ROI +\${roi}%\n\` + \`🌍 \${deal.country}\n\` + \`🔗 \${deal.url}\`; await fetch(\`https://api.telegram.org/bot\${token}/sendMessage\`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ chat_id: chatId, text: msg }) }); } } export default { async scheduled(event, env, ctx) { const brands = ['Carhartt','Nike','Adidas']; const allDeals = []; for (const brand of brands) { for (const c of VINTED_COUNTRIES) { const deals = await searchVinted(c.domain, brand, 0, 20, false); allDeals.push(...deals); } } allDeals.sort((a,b) => (b.price*2.8-b.price)/b.price - (a.price*2.8-a.price)/a.price); await sendTelegram(env.TELEGRAM_TOKEN, env.CHAT_ID, allDeals); }, async fetch(request, env) { const url = new URL(request.url); if (url.pathname === '/deals') { const brand = url.searchParams.get('brand') || 'Carhartt'; const min = url.searchParams.get('min') || 0; const max = url.searchParams.get('max') || 20; const rare = url.searchParams.get('rare') === '1'; const allDeals = []; for (const c of VINTED_COUNTRIES) { const deals = await searchVinted(c.domain, brand, min, max, rare); allDeals.push(...deals); } return new Response(JSON.stringify(allDeals), { headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' } }); } return new Response('DRIP.DEALS Worker Online', { status: 200 }); } };
